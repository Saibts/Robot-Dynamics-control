7. MANUAL TASK INJECTION VIA CLI
================================================================================
# Publish a single custom task to the coordinator at any time:
# Format: "TASK_NAME,PRIORITY(1=Urgent, 2=High, 3=Normal),STATION_NAME"
source /opt/ros/jazzy/setup.bash
ros2 topic pub --once /submit_task std_msgs/msg/String "{data: 'TASK_VIP_01,1,STORAGE_STATION_B'}"


================================================================================
8. DIRECT ROS 2 ACTION COMMANDS VIA CLI
================================================================================
# Send goal directly to TurtleBot3 + OpenManipulator:
source /opt/ros/jazzy/setup.bash
source ~/Desktop/RDC_project/2_Gazebo_And_RViz_Simulation/ros2_ws/install/setup.bash
ros2 action send_goal --feedback /navigate_and_pick multi_robot_interfaces/action/NavigateAndPick \
  "{target_station_id: 'STORAGE_STATION_B', pickup_coordinates: [1.45, 1.20, 0.25], priority_level: 1}"

# Send goal directly to UR5 Industrial Arm:
source /opt/ros/jazzy/setup.bash
source ~/Desktop/RDC_project/2_Gazebo_And_RViz_Simulation/ros2_ws/install/setup.bash
